    ```html
        import { createClient } from '@supabase/supabase-js';

        // ── Hardcoded demo values (D-06) ────────────
        // Replace with your Supabase project credentials
        const SUPABASE_URL = 'https://your-project.supabase.co';
        const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...';

        // ── State ────────────────────────────────────
        const requests = [];        // { userId, transcript[], timestamp }
        let activeUserId = null;    // currently accepted request
        let isConnected = false;

        // ── DOM refs ────────────────────────────────
        const statusDot = document.getElementById('status-dot');
        const statusText = document.getElementById('status-text');
        const requestsContainer = document.getElementById('requests-container');
        const chatHeader = document.getElementById('chat-header');
        const chatMessages = document.getElementById('chat-messages');
        const agentTextarea = document.getElementById('agent-textarea');
        const sendBtn = document.getElementById('send-btn');

        // ── Supabase connection ─────────────────────
        const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
        const channel = supabase.channel('support-queue', {
          config: { broadcast: { self: true } }, // Agent console sees its own events
        });

        function setConnectionStatus(status) {
          statusDot.className = 'dot ' + status;
          if (status === 'connected') {
            statusText.textContent = 'Connected ✓';
            isConnected = true;
          } else if (status === 'connecting') {
            statusText.textContent = 'Connecting...';
            isConnected = false;
          } else {
            statusText.textContent = 'Disconnected';
            isConnected = false;
          }
        }

        // ── Event handlers ──────────────────────────
        channel
          .on('broadcast', { event: 'handoff_request' }, (payload) => {
            // payload: { userId, transcript[], timestamp }
            if (!requests.find(r => r.userId === payload.userId)) {
              requests.push({
                userId: payload.userId,
                transcript: payload.transcript || [],
                timestamp: payload.timestamp,
              });
              renderRequestList();
            }
          })
          .on('broadcast', { event: 'handoff_accepted' }, (payload) => {
            // Update status if this console accepted
            const card = document.querySelector(`[data-user-id="${payload.userId}"]`);
            if (card) card.classList.add('accepted');
          })
          .subscribe((status) => {
            if (status === 'SUBSCRIBED') {
              setConnectionStatus('connected');
            } else {
              setConnectionStatus('disconnected');
            }
          });

        // ── Render functions ─────────────────────────
        function renderRequestList() {
          if (requests.length === 0) {
            requestsContainer.innerHTML = '<div class="empty-state">Waiting for handoff requests...</div>';
            return;
          }
          requestsContainer.innerHTML = '';
          for (const req of requests) {
            const card = document.createElement('div');
            card.className = 'request-card' + (req.userId === activeUserId ? ' accepted' : '');
            card.dataset.userId = req.userId;

            const userIdEl = document.createElement('div');
            userIdEl.className = 'user-id';
            userIdEl.textContent = req.userId;

            const preview = document.createElement('div');
            preview.className = 'preview';
            const lastMsg = req.transcript && req.transcript.length > 0
              ? req.transcript[req.transcript.length - 1].text || ''
              : '';
            preview.textContent = lastMsg || '(empty transcript)';

            card.appendChild(userIdEl);
            card.appendChild(preview);

            if (req.userId !== activeUserId) {
              const acceptBtn = document.createElement('button');
              acceptBtn.className = 'accept-btn';
              acceptBtn.textContent = 'Accept';
              acceptBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                acceptRequest(req.userId);
              });
              card.appendChild(acceptBtn);
            }

            requestsContainer.appendChild(card);
          }
        }

        function renderChat() {
          chatMessages.innerHTML = '';
          if (!activeUserId) {
            chatMessages.innerHTML = '<div class="chat-empty">Accept a handoff request to see the conversation transcript.</div>';
            return;
          }

          const req = requests.find(r => r.userId === activeUserId);
          if (!req || !req.transcript || req.transcript.length === 0) {
            chatMessages.innerHTML = '<div class="chat-empty">No messages in this conversation yet.</div>';
            return;
          }

          for (const msg of req.transcript) {
            const el = document.createElement('div');
            const role = msg.role === 'user' ? 'user' : 'agent';
            el.className = 'transcript-msg ' + role;
            el.textContent = msg.text || '';

            const meta = document.createElement('div');
            meta.className = 'meta';
            meta.textContent = msg.role === 'user' ? 'Customer' : 'AI Agent';
            el.appendChild(meta);

            chatMessages.appendChild(el);
          }
          chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        function addAgentMessage(text) {
          const el = document.createElement('div');
          el.className = 'transcript-msg agent';
          el.textContent = text;
          const meta = document.createElement('div');
          meta.className = 'meta';
          meta.textContent = 'You (Human Agent)';
          el.appendChild(meta);
          chatMessages.appendChild(el);
          chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        // ── Actions ─────────────────────────────────
        function acceptRequest(userId) {
          activeUserId = userId;
          chatHeader.textContent = 'Chat — ' + userId;
          renderChat();
          agentTextarea.disabled = false;
          sendBtn.disabled = false;
          agentTextarea.focus();

          // Send acceptance event
          channel.send({
            type: 'broadcast',
            event: 'handoff_accepted',
            payload: { userId, agentId: 'agent-1', timestamp: Date.now() },
          });

          renderRequestList();
        }

        function sendResponse() {
          const text = agentTextarea.value.trim();
          if (!text || !activeUserId) return;

          channel.send({
            type: 'broadcast',
            event: 'agent_message',
            payload: { userId: activeUserId, text, timestamp: Date.now() },
          });

          addAgentMessage(text);
          agentTextarea.value = '';
          agentTextarea.style.height = 'auto';
        }

        // ── Event bindings ──────────────────────────
        sendBtn.addEventListener('click', sendResponse);
        agentTextarea.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            sendResponse();
          }
        });
        agentTextarea.addEventListener('input', () => {
          agentTextarea.style.height = 'auto';
          agentTextarea.style.height = agentTextarea.scrollHeight + 'px';
        });

        // ── Initial state ──────────────────────────
        setConnectionStatus('connecting');
      </script>
    </body>
    </html>
    ```
